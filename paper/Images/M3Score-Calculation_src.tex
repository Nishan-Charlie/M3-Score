\node[font=\bfseries\small, text=cReal] at (0.12,2.55) {Real images \ $R$};
\node[font=\bfseries\small, text=cGen]  at (0.12,-2.28) {Generated images \ $G$};
\node[font=\scriptsize, text=black!60] at (0.12,-2.68) {$N$ images in each set};

\coordinate (inR) at (0.82,1.38);
\coordinate (inG) at (0.82,-1.52);

% ---- preprocess ----------------------------------------------------
\node[box, draw=black!45, fill=black!3] (prep) at (2.85,-0.10)
     {\textbf{Preprocess}\\[1pt]
      {\scriptsize resize to $224\!\times\!224$}\\[-1pt]
      {\scriptsize normalise}};
\draw[ar] (inR) -- ++(0.45,0) |- (prep.west);
\draw[ar] (inG) -- ++(0.45,0) |- (prep.west);

% ---- frozen encoder ------------------------------------------------
\node[box, draw=cFroz, fill=cFroz!6, text width=4.35cm] (enc) at (6.95,-0.10)
     {\textbf{\ding{100}\ Frozen encoder: RadioDINO-s16}\\[2pt]
      {\scriptsize ViT-S/16 $\cdot$ 12 blocks $\cdot$ $D\!=\!384$}\\[3pt]
      {\scriptsize Pre-trained on \textbf{838\,000 radiology}}\\[-1pt]
      {\scriptsize \textbf{images} (X-ray, CT and MRI) using}\\[-1pt]
      {\scriptsize DINOv2 self-supervision}\\[3pt]
      {\scriptsize\itshape All 12 blocks stay frozen ---}\\[-1pt]
      {\scriptsize\itshape nothing here is trained.}};
\draw[ar] (prep) -- (enc);

% ---- which blocks are read ----------------------------------------
\foreach \i in {1,...,12}{
  \pgfmathsetmacro{\x}{5.02 + (\i-1)*0.35}
  \node[draw=cFroz!55, fill=cFroz!16, minimum width=0.28cm,
        minimum height=0.28cm, inner sep=0pt, line width=0.4pt]
        at (\x,-2.42) {};
}
\foreach \i/\c in {4/cCov, 9/cMem, 12/cFid}{
  \pgfmathsetmacro{\x}{5.02 + (\i-1)*0.35}
  \node[draw=\c, fill=\c!35, minimum width=0.28cm, minimum height=0.28cm,
        inner sep=0pt, line width=1.1pt] at (\x,-2.42) {};
}
\node[font=\scriptsize, text=black!60] at (6.95,-2.80) {block 1 \ $\longrightarrow$ \ block 12};
\node[font=\scriptsize, align=center] at (7.35,-3.22)
     {read from \textbf{\textcolor{cCov}{L4}},
      \textbf{\textcolor{cMem}{L9}} and \textbf{\textcolor{cFid}{L12}}};
\draw[black!35, line width=0.6pt] (enc.south) -- (6.95,-2.24);

% ---- CLS features --------------------------------------------------
\node[box, draw=black!45, fill=black!3, text width=2.75cm] (cls) at (10.55,-0.10)
     {\textbf{CLS feature}\\[2pt]
      {\scriptsize one vector per image,}\\[-1pt]
      {\scriptsize per chosen block}\\[3pt]
      {\scriptsize $E_r^{(\ell)},E_g^{(\ell)}\!\in\!\mathbb{R}^{N\times384}$}};
\draw[ar] (enc) -- (cls);

% ---- the three questions ------------------------------------------
\node[box, draw=cFid, fill=cFid!7, text width=6.35cm, anchor=west]
     (qfid) at (13.1,2.02)
     {\textbf{\textcolor{cFid}{1. Fidelity}} \ \ {\scriptsize block \textbf{L12}}\\[2pt]
      \emph{\scriptsize Do the generated images match the real ones}\\[-1pt]
      \emph{\scriptsize as a group?}\\[2pt]
      {\scriptsize Distance between the two sets of features}\\[-1pt]
      {\scriptsize (unbiased multi-bandwidth RBF MMD$^2$)}};

\node[box, draw=cMem, fill=cMem!7, text width=6.35cm, anchor=west]
     (qmem) at (13.1,-0.42)
     {\textbf{\textcolor{cMem}{2. Memorisation}} \ \ {\scriptsize block \textbf{L9}}\\[2pt]
      \emph{\scriptsize Has the model simply copied its training images?}\\[2pt]
      {\scriptsize How close each generated image lies to its}\\[-1pt]
      {\scriptsize nearest real image (1-NN distance)}};

\node[box, draw=cCov, fill=cCov!7, text width=6.35cm, anchor=west]
     (qcov) at (13.1,-2.86)
     {\textbf{\textcolor{cCov}{3. Coverage}} \ \ {\scriptsize block \textbf{L4}}
      \ {\scriptsize\itshape --- diagnostic only}\\[2pt]
      \emph{\scriptsize Does it produce the full variety of real images,}\\[-1pt]
      \emph{\scriptsize or only a few kinds?}\\[2pt]
      {\scriptsize $k$-NN precision and recall between the sets}};

\begin{scope}[on background layer]
  \node[rounded corners=5pt, draw=black!35, fill=black!2, line width=1pt,
        fit=(qfid)(qmem)(qcov), inner sep=10pt] (qbox) {};
\end{scope}
\node[font=\bfseries\small, anchor=south west]
      at ([shift={(1pt,3pt)}]qbox.north west)
      {Three separate questions --- each answered at its own block};

\draw[ar] (cls.east) -- ++(0.50,0) |- (qfid.west);
\draw[ar] (cls.east) -- ++(0.50,0) |- (qmem.west);
\draw[ar] (cls.east) -- ++(0.50,0) |- (qcov.west);

% =====================================================================
% ROW 2 : post-hoc loop | how the score is formed | report | evidence
% =====================================================================

% ---- post-hoc validation (with return path) ------------------------
\node[box, draw=cVal, fill=cVal!5, dashed, text width=5.15cm, anchor=north]
     (val) at (3.05,-4.60)
     {\textbf{\textcolor{cVal}{Post-hoc check}}\\[1pt]
      {\scriptsize\itshape does not feed the reported score}\\[3pt]
      {\scriptsize A CKA analysis of the same features asks}\\[-1pt]
      {\scriptsize whether the blocks fixed in advance really}\\[-1pt]
      {\scriptsize are the non-redundant ones.}\\[3pt]
      {\scriptsize It \textbf{confirmed L12}. Were a future check to}\\[-1pt]
      {\scriptsize disagree, the layer choice is revised
       \emph{before}}\\[-1pt]
      {\scriptsize the next evaluation --- the loop below.}};

% features in  (down the clear lane at y = -3.95)
\draw[arv] (cls.south) -- (10.55,-3.95) -- (2.20,-3.95) -- (2.20,-4.60);
% and the loop back out to the layer choice
\draw[arv] (4.60,-4.60) -- (4.60,-2.42) -- (4.78,-2.42);
\node[font=\scriptsize\itshape, text=cVal, anchor=west] at (4.75,-3.72)
      {revise the layer choice};

% ---- how the score is formed ---------------------------------------
\node[box, draw=cRep, fill=cRep!7, text width=5.25cm, anchor=north]
     (comb) at (9.55,-4.60)
     {\textbf{\textcolor{cRep}{How the M3-Score is formed}}\\[3pt]
      {\scriptsize The headline \textbf{M3-Score is the fidelity}}\\[-1pt]
      {\scriptsize \textbf{number from L12 on its own}.}\\[4pt]
      {\scriptsize The three axes are \textbf{never averaged}.}\\[-1pt]
      {\scriptsize A single mean would hide \emph{which} of the}\\[-1pt]
      {\scriptsize three failed, so memorisation and}\\[-1pt]
      {\scriptsize coverage are reported \textbf{beside} it.}};

\draw[ar] (qbox.south) -- ++(0,-0.45) -| (comb.north);

% ---- report ---------------------------------------------------------
\node[box, draw=cRep, fill=cRep!4, text width=4.35cm, anchor=north]
     (rep) at (15.35,-4.60)
     {\textbf{\textcolor{cRep}{M3-Score report}}\\[3pt]
      {\scriptsize $\bullet$\ \ \textbf{M3} $=$ L12 fidelity MMD$^2$,}\\[-1pt]
      {\scriptsize \ \ \ with permutation $p$, bootstrap}\\[-1pt]
      {\scriptsize \ \ \ 95\% CI and separation $Z$}\\[2pt]
      {\scriptsize $\bullet$\ \ \textbf{memorisation rate}}\\[2pt]
      {\scriptsize $\bullet$\ \ \textbf{coverage} (diagnostic)}};
\draw[ar] (comb) -- (rep);
